// ── TREE TOGGLE ──
function toggle(row) {
  const chev = row.querySelector('.tree-chev');
  const children = row.nextElementSibling;
  if (!children || !children.classList.contains('tree-children')) return;
  children.classList.toggle('hidden');
  if (chev) chev.classList.toggle('open');
}
 
// ── SELECT NODE ──
function selectNode(row) {
  document.querySelectorAll('.tree-row.selected').forEach(r => r.classList.remove('selected'));
  row.classList.add('selected');
}

function closeModal() { document.getElementById('modal').classList.remove('open'); }

  if (e.target === this) closeModal();
});

  menu.style.top = e.clientY + 'px';
  menu.classList.add('open');
}

// ── AI SEND ──
function sendMsg() {
  const ta = document.getElementById('ai-textarea');
  const text = ta.value.trim();
  if (!text) return;
  const chat = document.getElementById('ai-chat');
  // user msg
  const userDiv = document.createElement('div');
  userDiv.className = 'ai-msg';
  userDiv.innerHTML = `<div class="ai-msg-from">YOU</div><div class="ai-bubble user">${text}</div>`;
  chat.appendChild(userDiv);
  ta.value = '';
  // thinking
  const thinkDiv = document.createElement('div');
  thinkDiv.className = 'ai-msg';
  thinkDiv.innerHTML = `<div class="ai-msg-from">ASSISTANT</div><div class="ai-bubble assistant"><div class="ai-thinking"><span></span><span></span><span></span></div></div>`;
  chat.appendChild(thinkDiv);
  chat.scrollTop = chat.scrollHeight;
  setTimeout(() => {
    thinkDiv.querySelector('.ai-bubble').innerHTML = `I'm analyzing your question about <em>"${text}"</em>. Based on your current session context and the query in your editor, here's what I found — the <code>sys.dm_exec_sessions</code> DMV and related views give you full visibility into this.`;
    chat.scrollTop = chat.scrollHeight;
  }, 1200);
}
 
function sendSuggestion(el) {
  const text = el.textContent.replace('→ ', '');
  document.getElementById('ai-textarea').value = text;
  el.closest('.ai-suggestions').remove();
  sendMsg();
}
 
function aiKeydown(e) {

}

    setTimeout(() => btn.textContent = '⧉', 1500);
  });
}
